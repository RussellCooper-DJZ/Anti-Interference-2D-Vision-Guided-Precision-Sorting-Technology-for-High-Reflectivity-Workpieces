## 3. 相机驱动实现

### 3.1 GigE相机基础驱动

```python
# src/camera/gige_camera.py
import numpy as np
from pypylon import pylon
import cv2
import time
from typing import Optional, Tuple

class GigECamera:
    """GigE工业相机驱动"""

    def __init__(self, config: dict):
        self.config = config
        self.camera: Optional[pylon.InstantCamera] = None
        self.converter = None
        self.is_open = False

    def open(self):
        """打开相机"""
        # 获取相机设备工厂
        tlf = pylon.TlFactory.GetInstance()

        # 获取所有设备
        devices = tlf.EnumerateDevices()

        if len(devices) == 0:
            raise RuntimeError("未找到相机设备!")

        # 查找第一个GigE相机
        for device in devices:
            if device.GetDeviceClass() == 'BaslerGigE':
                self.camera = pylon.InstantCamera(tlf.CreateDevice(device))
                break

        if self.camera is None:
            self.camera = pylon.InstantCamera(tlf.CreateDevice(devices[0]))

        # 打开相机
        self.camera.Open()

        # 配置相机参数
        self._configure_camera()

        # 创建图像转换器
        self.converter = pylon.ImageFormatConverter()
        self.converter.OutputPixelFormat = pylon.PixelType_Mono8
        self.converter.OutputBitAlignment = pylon.OutputBitAlignment_MsbRight

        self.is_open = True
        print(f"相机已打开: {self.camera.GetDeviceInfo().GetModelName()}")

    def _configure_camera(self):
        """配置相机参数"""
        # 获取节点映射
        nodemap = self.camera.GetNodeMap()

        # 设置分辨率
        width = self.config.get('resolution', {}).get('width', 4096)
        height = self.config.get('resolution', {}).get('height', 3000)

        # 设置AOI (Area of Interest)
        pylon.GuiFeatureNode('Width', nodemap).SetValue(width)
        pylon.GuiFeatureNode('Height', nodemap).SetValue(height)

        # 设置曝光
        exposure_time = self.config.get('exposure', {}).get('time', 10000)
        pylon.GuiFeatureNode('ExposureTime', nodemap).SetValue(exposure_time)

        # 设置增益
        gain = self.config.get('gain', {}).get('value', 1.0)
        pylon.GuiFeatureNode('Gain', nodemap).SetValue(gain)

        # 设置帧率
        fps = self.config.get('fps', 30)
        pylon.GuiFeatureNode('AcquisitionFrameRate', nodemap).SetValue(fps)

        # 设置触发模式
        trigger_mode = self.config.get('trigger', {}).get('mode', 'Off')
        pylon.GuiFeatureNode('TriggerMode', nodemap).SetValue(trigger_mode)

        if trigger_mode == 'On':
            trigger_source = self.config.get('trigger', {}).get('source', 'Software')
            pylon.GuiFeatureNode('TriggerSource', nodemap).SetValue(trigger_source)

    def grab(self, num_images: int = 1) -> np.ndarray:
        """
        采集图像

        Args:
            num_images: 采集图像数量

        Returns:
            image: 单张图像或图像列表
        """
        if not self.is_open:
            raise RuntimeError("相机未打开!")

        # 开始采集
        self.camera.StartGrabbing()

        images = []
        grab_result = self.camera.RetrieveResult(
            5000,  # 超时时间ms
            pylon.GrabStrategy_LatestImageOnly
        )

        try:
            if grab_result.GrabSucceeded():
                image = self.converter.Convert(grab_result)
                img_array = image.GetArray()
                images.append(img_array.copy())

            while len(images) < num_images and self.camera.IsGrabbing():
                grab_result = self.camera.RetrieveResult(5000, pylon.GrabStrategy_LatestImageOnly)
                if grab_result.GrabSucceeded():
                    image = self.converter.Convert(grab_result)
                    images.append(image.GetArray().copy())

        finally:
            grab_result.Release()

        if num_images == 1:
            return images[0] if images else None
        return images

    def grab_one(self) -> np.ndarray:
        """采集单张图像"""
        return self.grab(num_images=1)

    def set_exposure(self, exposure_us: float):
        """设置曝光时间"""
        if self.is_open:
            nodemap = self.camera.GetNodeMap()
            pylon.GuiFeatureNode('ExposureTime', nodemap).SetValue(exposure_us)

    def trigger(self):
        """软件触发"""
        if self.is_open:
            nodemap = self.camera.GetNodeMap()
            pylon.GuiFeatureNode('TriggerSoftware', nodemap).Execute()

    def close(self):
        """关闭相机"""
        if self.camera and self.is_open:
            self.camera.StopGrabbing()
            self.camera.Close()
            self.is_open = False
            print("相机已关闭")

    def __enter__(self):
        self.open()
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        self.close()
```

### 3.2 偏振光图像采集

```python
# src/camera/polarized_capture.py
import numpy as np
import cv2
import threading
import time
from typing import Optional, Tuple, List

class PolarizedImage:
    """偏振光图像数据结构"""

    def __init__(self, rgb: np.ndarray, intensity: np.ndarray,
                 degree: np.ndarray, angle: np.ndarray):
        """
        Args:
            rgb: RGB图像 (H, W, 3)
            intensity: 光强图像 (H, W)
            degree: 偏振度图像 (H, W)
            angle: 偏振角图像 (H, W)
        """
        self.rgb = rgb
        self.intensity = intensity
        self.degree = degree
        self.angle = angle
        self.timestamp = time.time()

class PolarizedCamera:
    """偏振光相机采集系统"""

    def __init__(self, camera, config: dict):
        """
        Args:
            camera: GigECamera实例
            config: 配置字典
        """
        self.camera = camera
        self.config = config
        self.is_running = False
        self.frame_buffer = None
        self.lock = threading.Lock()
        self.thread = None

        # 偏振片角度配置
        self.polarizer_angles = [0, 45, 90, 135]  # 四个偏振角度

    def capture_with_polarization(self) -> Optional[PolarizedImage]:
        """
        采集偏振光图像

        偏振相机通常有四个方向的偏振片，一次曝光可获取四个偏振角度的图像
        或者使用旋转偏振片多次采集

        Returns:
            PolarizedImage: 偏振光图像数据结构
        """
        if not self.camera.is_open:
            raise RuntimeError("相机未打开!")

        # 方式1: 如果是专业偏振相机，一次曝光获取所有偏振通道
        # 方式2: 普通相机需要旋转偏振片多次采集

        # 这里实现方式2的演示
        rgb_images = []
        for angle in self.polarizer_angles:
            # 设置偏振片角度（实际需要控制电机）
            # self.set_polarizer_angle(angle)

            # 等待偏振片稳定
            time.sleep(0.1)

            # 采集图像
            img = self.camera.grab_one()
            rgb_images.append(img)

        # 计算偏振参数
        intensity, degree, angle = self._compute_polarization(rgb_images)

        # 转换为RGB（如果是灰度相机）
        if len(rgb_images[0].shape) == 2:
            rgb = cv2.cvtColor(rgb_images[0], cv2.COLOR_GRAY2RGB)
        else:
            rgb = rgb_images[0]

        return PolarizedImage(
            rgb=rgb,
            intensity=intensity,
            degree=degree,
            angle=angle
        )

    def _compute_polarization(self, images: List[np.ndarray]) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        计算偏振参数

        使用Stokes向量方法计算偏振度和偏振角

        Args:
            images: 四个角度的图像列表 [I0, I45, I90, I135]

        Returns:
            intensity: 光强
            degree: 偏振度
            angle: 偏振角
        """
        # 确保是float类型
        I0 = images[0].astype(np.float32)
        I45 = images[1].astype(np.float32)
        I90 = images[2].astype(np.float32)
        I135 = images[3].astype(np.float32)

        # 计算Stokes向量分量
        S0 = (I0 + I90) / 2  # 总强度
        S1 = (I0 - I90) / 2   # 水平-垂直偏振分量
        S2 = (I45 - I135) / 2  # 对角线偏振分量

        # 总强度（光强）
        intensity = S0

        # 偏振度
        with np.errstate(divide='ignore', invalid='ignore'):
            DoLP = np.sqrt(S1**2 + S2**2) / (S0 + 1e-10)
            DoLP = np.clip(DoLP, 0, 1)

        # 偏振角 (0到pi)
        AoLP = 0.5 * np.arctan2(S2, S1)
        # 转换为度数
        AoLP_deg = np.rad2deg(AoLP)

        return intensity, DoLP, AoLP_deg

    def process_image(self, polarized_img: PolarizedImage,
                     highlight_threshold: float = 0.7,
                     suppress_ratio: float = 0.5) -> np.ndarray:
        """
        处理偏振图像，抑制高光

        Args:
            polarized_img: 偏振图像
            highlight_threshold: 高光阈值（偏振度高于此值认为是镜面反射）
            suppress_ratio: 抑制比例

        Returns:
            processed: 处理后的灰度图像
        """
        # 使用偏振度识别高光区域
        highlight_mask = (polarized_img.degree > highlight_threshold).astype(np.float32)

        # 抑制高光
        suppressed = polarized_img.intensity * (1 - suppress_ratio * highlight_mask)

        # 局部对比度增强
        clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        enhanced = clahe.apply(suppressed.astype(np.uint8))

        return enhanced

    def start_continuous_capture(self):
        """开始连续采集"""
        self.is_running = True
        self.thread = threading.Thread(target=self._capture_loop)
        self.thread.daemon = True
        self.thread.start()

    def _capture_loop(self):
        """连续采集循环"""
        while self.is_running:
            try:
                polarized_img = self.capture_with_polarization()
                processed = self.process_image(polarized_img)

                with self.lock:
                    self.frame_buffer = processed.copy()

            except Exception as e:
                print(f"采集错误: {e}")
                time.sleep(0.1)

    def get_frame(self) -> Optional[np.ndarray]:
        """获取当前帧"""
        with self.lock:
            if self.frame_buffer is not None:
                return self.frame_buffer.copy()
        return None

    def stop(self):
        """停止采集"""
        self.is_running = False
        if self.thread:
            self.thread.join(timeout=2)
```

### 3.3 HDR成像实现

```python
# src/camera/hdr_capture.py
import numpy as np
import cv2
import time
from typing import List, Tuple, Optional

class HDRImaging:
    """高动态范围成像"""

    def __init__(self, camera, config: dict):
        """
        Args:
            camera: GigECamera实例
            config: HDR配置
        """
        self.camera = camera
        self.config = config
        self.exposure_times = config.get('exposure_times',
                                         [1/500, 1/250, 1/125, 1/60, 1/30])

    def capture_hdr_sequence(self) -> List[Tuple[np.ndarray, float]]:
        """
        采集HDR图像序列

        Returns:
            sequences: [(image, exposure_time), ...] 列表
        """
        sequences = []

        for exp_time in self.exposure_times:
            # 设置曝光时间（转换为微秒）
            exp_us = int(1.0 / exp_time * 1e6) if exp_time < 1 else int(exp_time * 1e6)
            self.camera.set_exposure(exp_us)

            # 等待相机稳定
            time.sleep(0.05)

            # 采集图像
            img = self.camera.grab_one()
            sequences.append((img, exp_time))

        return sequences

    def merge_mertens(self, sequences: List[Tuple[np.ndarray, float]]) -> np.ndarray:
        """
        使用Mertens方法融合HDR

        Args:
            sequences: [(image, exposure_time), ...]

        Returns:
            hdr: 融合后的HDR图像 [0, 1]
        """
        images = [seq[0] for seq in sequences]

        # 创建融合器
        merge = cv2.createMergeMertens()

        # 处理图像格式（需要32位浮点）
        float_images = []
        for img in images:
            if img.dtype == np.uint8:
                img_float = img.astype(np.float32) / 255.0
            else:
                img_float = img.astype(np.float32)
            float_images.append(img_float)

        # 融合
        merged = merge.process(float_images)

        return np.clip(merged, 0, 1)

    def merge_debevec(self, sequences: List[Tuple[np.ndarray, float]],
                     gamma: float = 1.0) -> np.ndarray:
        """
        使用 Debevec 方法融合HDR

        Args:
            sequences: [(image, exposure_time), ...]
            gamma: gamma校正值

        Returns:
            hdr: HDR图像
        """
        images = []
        times = []

        for img, exp_time in sequences:
            if img.dtype == np.uint8:
                images.append(img.astype(np.float32) / 255.0)
            else:
                images.append(img.astype(np.float32))
            times.append(exp_time)

        images = np.array(images)
        times = np.array(times, dtype=np.float32)

        # 估计相机响应函数
        calibrate = cv2.createCalibrateDebevec()
        response = calibrate.process(images, times)

        # 融合HDR
        merge_debevec = cv2.createMergeDebevec()
        hdr = merge_debevec.process(images, times, response)

        # Gamma校正
        if gamma != 1.0:
            hdr = np.power(hdr, 1.0 / gamma)

        return hdr

    def tonemap(self, hdr: np.ndarray,
                method: str = 'Drago') -> np.ndarray:
        """
        色调映射，将HDR转为可显示的LDR

        Args:
            hdr: HDR图像 [0, 1] 或更大
            method: 'Drago', 'Reinhard', 'Mantiuk', 'Ferry'

        Returns:
            ldr: 色调映射后的图像 [0, 255] uint8
        """
        if method == 'Drago':
            tonemap = cv2.createTonemapDrago(1.0, 0.3)
        elif method == 'Reinhard':
            tonemap = cv2.createTonemapReinhard(1.0, 0.5, 0.0, 0.0)
        elif method == 'Mantiuk':
            tonemap = cv2.createTonemapMantiuk(2.2, 0.85, 1.2)
        elif method == 'Ferry':
            tonemap = cv2.createTonemap Ferrywal(1.0)
        else:
            raise ValueError(f"未知的色调映射方法: {method}")

        # 映射
        ldr = tonemap.process(hdr)

        # 归一化到[0, 255]
        ldr = np.clip(ldr * 255, 0, 255).astype(np.uint8)

        return ldr

    def capture_and_fuse(self) -> Tuple[np.ndarray, np.ndarray]:
        """
        采集HDR并融合

        Returns:
            fused: 融合后的图像 [0, 1]
            ldr: 色调映射后的可显示图像 [0, 255]
        """
        # 采集序列
        sequences = self.capture_hdr_sequence()

        # Mertens融合
        fused = self.merge_mertens(sequences)

        # 色调映射
        ldr = self.tonemap(fused)

        return fused, ldr
```

```python
import cv2
from typing import List, Dict, Optional, Tuple

class YOLODetector:
    """YOLOv8缺陷检测器"""

    # 缺陷类别名称

        'patches',         # 斑块

    @torch.no_grad()
    def detect(self, image: np.ndarray) -> List[DefectDetectionResult]:
        """
        检测缺陷

        Args:
            image: RGB图像 (H, W, 3), uint8 或 float32

        Returns:
            results: 检测结果列表
        """
        if self.use_trt:
            return self._detect_trt(image)
        else:
            return self._detect_yolo(image)

    def _detect_yolo(self, image: np.ndarray) -> List[DefectDetectionResult]:

        # 推理
        results = self.model.predict(

            conf=self.conf_threshold,
            iou=self.iou_threshold,
            verbose=False,
            device=self.device
        )

        # 解析结果
        detections = []
        for result in results:
            boxes = result.boxes
            for box in boxes:
                x1, y1, x2, y2 = box.xyxy[0].cpu().numpy()
                conf = float(box.conf[0])
                cls = int(box.cls[0])
                name = self.CLASS_NAMES[cls] if cls < len(self.CLASS_NAMES) else f"class_{cls}"

                detections.append(DefectDetectionResult(
                    bbox=[float(x1), float(y1), float(x2), float(y2)],
                    conf=conf,
                    cls=cls,
                    name=name
                ))

        return detections

    def _detect_trt(self, image: np.ndarray) -> List[DefectDetectionResult]:
        """使用TensorRT模型检测"""
        # TensorRT推理实现
        # 预处理
        input_tensor = self._preprocess(image)

        # 推理
        outputs = self.trt_model.inference(input_tensor)

        # 后处理
        detections = self._postprocess_trt(outputs)

        return detections

    def _preprocess(self, image: np.ndarray) -> torch.Tensor:
        """预处理"""
        # 调整大小
        img = cv2.resize(image, (640, 640))

        # 归一化
        img = img.astype(np.float32) / 255.0

        # HWC to CHW
        img = np.transpose(img, (2, 0, 1))

        # NCHW
        img = np.expand_dims(img, 0)

        return torch.from_numpy(img).cuda()

    def _postprocess_trt(self, outputs: np.ndarray) -> List[DefectDetectionResult]:
        """后处理TensorRT输出"""
        # YOLO输出格式: (1, 84, 8400) [obj, x, y, w, h, cls...]
        detections = []

        outputs = outputs[0]  # (84, 8400)
        outputs = outputs.T   # (8400, 84)

        for det in outputs:
            obj_conf = det[4]
            if obj_conf < self.conf_threshold:
                continue

            cls_confs = det[5:]
            cls_id = np.argmax(cls_confs)
            cls_conf = cls_confs[cls_id]

            if cls_conf < self.conf_threshold:
                continue

            # 解析坐标
            x, y, w, h = det[:4]
            x1 = x - w / 2
            y1 = y - h / 2
            x2 = x + w / 2
            y2 = y + h / 2

            detections.append(DefectDetectionResult(
                bbox=[float(x1), float(y1), float(x2), float(y2)],
                conf=float(obj_conf * cls_conf),
                cls=int(cls_id),
                name=self.CLASS_NAMES[cls_id]
            ))

        # NMS
        detections = self._nms(detections)

        return detections

    def _nms(self, detections: List[DefectDetectionResult],
             iou_threshold: float = 0.4) -> List[DefectDetectionResult]:
        """非极大值抑制"""
        if not detections:
            return []

        boxes = np.array([d.bbox for d in detections])
        scores = np.array([d.conf for d in detections])

        # 转换为xywh格式
        x1 = boxes[:, 0]
        y1 = boxes[:, 1]
        x2 = boxes[:, 2]
        y2 = boxes[:, 3]

        areas = (x2 - x1) * (y2 - y1)

        # 按分数排序
        order = scores.argsort()[::-1]

        keep = []
        while order.size > 0:
            i = order[0]
            keep.append(i)

            # 计算IOU
            xx1 = np.maximum(x1[i], x1[order[1:]])
            yy1 = np.maximum(y1[i], y1[order[1:]])
            xx2 = np.minimum(x2[i], x2[order[1:]])
            yy2 = np.minimum(y2[i], y2[order[1:]])

            w = np.maximum(0.0, xx2 - xx1)
            h = np.maximum(0.0, yy2 - yy1)
            inter = w * h

            iou = inter / (areas[i] + areas[order[1:]] - inter)

            # 保留IOU小于阈值的
            inds = np.where(iou <= iou_threshold)[0]
            order = order[inds + 1]

        return [detections[i] for i in keep]

    def draw_detections(self, image: np.ndarray,
                       detections: List[DefectDetectionResult]) -> np.ndarray:
        """绘制检测结果"""
        result_img = image.copy()

        for det in detections:
            x1, y1, x2, y2 = [int(v) for v in det.bbox]

            # 颜色
            color = self._get_class_color(det.cls)

            # 绘制边界框
            cv2.rectangle(result_img, (x1, y1), (x2, y2), color, 2)

            # 绘制标签
            label = f"{det.name}: {det.conf:.2f}"
            label_size, _ = cv2.getTextSize(label, cv2.FONT_HERSHEY_SIMPLEX, 0.5, 1)
            cv2.rectangle(result_img, (x1, y1 - label_size[1] - 5),
                         (x1 + label_size[0], y1), color, -1)
            cv2.putText(result_img, label, (x1, y1 - 5),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

        return result_img

    def _get_class_color(self, cls: int) -> Tuple[int, int, int]:
        """获取类别颜色"""
        colors = [
            (255, 0, 0),    # 裂纹 - 红色
            (0, 255, 0),    # 夹杂 - 绿色
            (0, 0, 255),    # 斑块 - 蓝色
            (255, 255, 0),  # 凹坑 - 青色
            (255, 0, 255),  # 氧化皮 - 紫色
            (0, 255, 255)  # 划痕 - 黄色
        ]
        return colors[cls % len(colors)]
```

```python
import tensorrt as trt
import torch
import numpy as np
import pycuda.driver as cuda
import pycuda.autoinit

class TensorRTInfer:
    """TensorRT推理引擎"""

    def __init__(self, engine_path: str, max_batch_size: int = 8):
        """
        Args:
            engine_path: TensorRT引擎路径
            max_batch_size: 最大batch size
        """
        self.logger = trt.Logger(trt.Logger.WARNING)
        self.runtime = trt.Runtime(self.logger)
        self.engine = self.runtime.deserialize_cuda_engine(engine_path)
        self.context = self.engine.create_execution_context()
        self.max_batch_size = max_batch_size

        # 获取输入输出索引
        self.input_idx = self.engine.get_binding_index('images')
        self.output_idx = self.engine.get_binding_index('output')

        # 获取形状
        self.input_shape = self.engine.get_binding_shape(self.input_idx)
        self.output_shape = self.engine.get_binding_shape(self.output_idx)

        # 分配GPU内存
        self.input_size = max_batch_size * self._get_size_from_shape(self.input_shape)
        self.output_size = max_batch_size * self._get_size_from_shape(self.output_shape)

        self.d_input = cuda.mem_alloc(self.input_size)
        self.d_output = cuda.mem_alloc(self.output_size)

        # 绑定流
        self.stream = cuda.Stream()

    def _get_size_from_shape(self, shape):
        """计算tensor大小"""
        size = 1
        for dim in shape:
            size *= dim
        return size

    def inference(self, inputs: np.ndarray) -> np.ndarray:
        """
        执行推理

        Args:
            inputs: 输入数据 (N, C, H, W), float32

        Returns:
            outputs: 输出数据
        """
        batch_size = inputs.shape[0]

        # 设置动态batch
        self.context.set_binding_shape(self.input_idx, inputs.shape)

        # 传输数据到GPU
        cuda.memcpy_htod_async(self.d_input, inputs, self.stream)

        # 执行推理
        self.context.execute_async_v2(
            bindings=[int(self.d_input), int(self.d_output)],
            stream_handle=self.stream.handle
        )

        # 传输结果回CPU
        output = np.empty(self.output_shape, dtype=np.float32)
        cuda.memcpy_dtoh_async(output, self.d_output, self.stream)

        # 同步
        self.stream.synchronize()

        return output

    def __del__(self):
        """清理"""
        if hasattr(self, 'd_input'):
            self.d_input.free()
        if hasattr(self, 'd_output'):
            self.d_output.free()

class TensorRTBuilder:
    """TensorRT模型构建器"""

    def __init__(self, logger: trt.Logger = None):
        self.logger = logger or trt.Logger(trt.Logger.WARNING)
        self.builder = trt.Builder(self.logger)
        self.network = None
        self.config = None

    def build_engine(self, onnx_path: str, engine_path: str,
                   fp16: bool = True, int8: bool = False):
        """从ONNX构建TensorRT引擎"""
        # 创建网络
        network_flags = 1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
        self.network = self.builder.create_network(network_flags)

        # 创建builder配置
        self.config = self.builder.create_builder_config()
        self.config.max_workspace_size = 1 << 30  # 1GB

        # FP16优化
        if fp16 and self.builder.platform_has_fast_fp16:
            self.config.set_flag(trt.BuilderFlag.FP16)

        # INT8量化
        if int8:
            self.config.set_flag(trt.BuilderFlag.INT8)

        # 解析ONNX
        parser = trt.OnnxParser(self.network, self.logger)
        with open(onnx_path, 'rb') as f:
            parser.parse(f.read())

        # 构建引擎
        engine = self.builder.build_serialized_network(self.network, self.config)

        # 保存引擎
        with open(engine_path, 'wb') as f:

        print(f"TensorRT引擎已保存: {engine_path}")

            do_constant_folding=True,
            input_names=['images'],
            output_names=['output'],
            dynamic_axes={
                'images': {0: 'batch'},
                'output': {0: 'batch'}
            } if dynamic_batch else None
        )

        print(f"ONNX模型已导出: {output_path}")
```

```python
import numpy as np
from typing import Tuple, Optional

class CannyEdgeDetector:
    """Canny边缘检测器"""

    def __init__(self, config: dict):
        """
        Args:
            config: 配置字典
        """
        self.gaussian_ksize = config.get('gaussian_ksize', 5)
        self.gaussian_sigma = config.get('gaussian_sigma', 1.5)
        self.high_threshold_percentile = config.get('high_threshold_percentile', 85)
        self.low_threshold_ratio = config.get('low_threshold_ratio', 0.4)

    def detect(self, gray_image: np.ndarray) -> np.ndarray:
        """
        Canny边缘检测

        Args:
            gray_image: 灰度图像 (H, W)

        Returns:
            edges: 二值边缘图 (H, W)
        """
        # 高斯模糊
        blurred = cv2.GaussianBlur(
            gray_image,
            (self.gaussian_ksize, self.gaussian_ksize),
            self.gaussian_sigma
        )

        # 自动计算阈值
        low_th, high_th = self._compute_adaptive_threshold(blurred)

        # Canny边缘检测
        edges = cv2.Canny(blurred, low_th, high_th)

        return edges

    def _compute_adaptive_threshold(self, gray_image: np.ndarray) -> Tuple[int, int]:
        """计算自适应阈值"""
        # 计算梯度幅值
        sobelx = cv2.Sobel(gray_image, cv2.CV_64F, 1, 0, ksize=3)
        sobely = cv2.Sobel(gray_image, cv2.CV_64F, 0, 1, ksize=3)
        magnitude = np.sqrt(sobelx**2 + sobely**2)

        # 基于直方图计算阈值
        high_th = np.percentile(magnitude, self.high_threshold_percentile)
        low_th = high_th * self.low_threshold_ratio

        return int(low_th), int(high_th)

    def detect_with_params(self, gray_image: np.ndarray,
                          low_threshold: int,
                          high_threshold: int) -> np.ndarray:
        """使用指定阈值检测"""
        blurred = cv2.GaussianBlur(
            gray_image,
            (self.gaussian_ksize, self.gaussian_ksize),
            self.gaussian_sigma
        )

        edges = cv2.Canny(blurred, low_threshold, high_threshold)
        return edges
```

```python
# src/edge/adaptive_edge.py
import cv2
import numpy as np
from typing import Tuple, Optional, List

class AdaptiveEdgeDetector:
    """自适应边缘检测器"""

    def __init__(self, config: dict):
        self.config = config
        self.gaussian_ksize = config.get('gaussian_ksize', 5)
        self.gaussian_sigma = config.get('gaussian_sigma', 1.5)
        self.morph_close_iterations = config.get('morph_close_iterations', 2)
        self.min_contour_length = config.get('min_contour_length', 20)

    def detect(self, gray_image: np.ndarray,
               defect_regions: List[dict] = None) -> Tuple[np.ndarray, List]:
        """
        完整边缘检测流程

        Args:
            gray_image: 灰度图像
            defect_regions: 缺陷区域列表，用于排除干扰

        Returns:
            edges: 边缘二值图
            contours: 轮廓列表
        """
        # 1. 去噪
        denoised = cv2.fastNlMeansDenoising(gray_image, h=10)

        # 2. CLAHE增强
        clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        enhanced = clahe.apply(denoised)

        # 3. 边缘检测
        edges = self._canny_adaptive(enhanced)

        # 4. 形态学闭操作
        edges = self._morphology_close(edges)

        # 5. 去除缺陷区域干扰
        if defect_regions:
            edges = self._mask_defect_regions(edges, defect_regions)

        # 6. 提取轮廓
        contours, _ = cv2.findContours(
            edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        # 7. 过滤短轮廓
        filtered_contours = self._filter_contours(contours)

        return edges, filtered_contours

    def _canny_adaptive(self, gray_image: np.ndarray) -> np.ndarray:
        """自适应Canny"""
        # 计算梯度
        sobelx = cv2.Sobel(gray_image, cv2.CV_64F, 1, 0, ksize=3)
        sobely = cv2.Sobel(gray_image, cv2.CV_64F, 0, 1, ksize=3)
        magnitude = np.sqrt(sobelx**2 + sobely**2)

        # 计算局部方差
        local_mean = cv2.blur(gray_image, (15, 15))
        local_var = cv2.blur((gray_image - local_mean) ** 2, (15, 15))

        # 自适应阈值
        high_th = np.percentile(magnitude, 85)
        low_th = high_th * 0.4

        # 根据局部方差调整
        variance_threshold = 50
        high_th = np.where(local_var > variance_threshold, high_th * 1.2, high_th)

        # 高斯模糊
        blurred = cv2.GaussianBlur(gray_image,
                                   (self.gaussian_ksize, self.gaussian_ksize),
                                   self.gaussian_sigma)

        # Canny
        edges = cv2.Canny(blurred, int(low_th), int(high_th))

        return edges

    def _morphology_close(self, edges: np.ndarray) -> np.ndarray:
        """形态学闭操作连接断点"""
        kernel = np.ones((3, 3), np.uint8)
        closed = cv2.morphologyEx(
            edges,
            cv2.MORPH_CLOSE,
            kernel,
            iterations=self.morph_close_iterations
        )
        return closed

    def _mask_defect_regions(self, edges: np.ndarray,
                            defect_regions: List[dict]) -> np.ndarray:
        """去除缺陷区域的边缘"""
        mask = np.ones_like(edges, dtype=np.uint8)

        for region in defect_regions:
            x1, y1, x2, y2 = [int(v) for v in region['bbox']]
            margin = 20
            x1 = max(0, x1 - margin)
            y1 = max(0, y1 - margin)
            x2 = min(edges.shape[1], x2 + margin)
            y2 = min(edges.shape[0], y2 + margin)
            mask[y1:y2, x1:x2] = 0

        return edges * mask

    def _filter_contours(self, contours) -> List:
        """过滤短轮廓"""
        filtered = []
        for contour in contours:
            length = cv2.arcLength(contour, True)
            if length >= self.min_contour_length:
                filtered.append(contour)
        return filtered

    def analyze_contour(self, contour) -> dict:
        """分析轮廓特征"""
        # 面积
        area = cv2.contourArea(contour)

        # 周长
        perimeter = cv2.arcLength(contour, True)

        # 边界框
        x, y, w, h = cv2.boundingRect(contour)

        # 最小外接矩形
        rect = cv2.minAreaRect(contour)

        # 圆形度
        circularity = 4 * np.pi * area / (perimeter ** 2) if perimeter > 0 else 0

        return {
            'area': area,
            'perimeter': perimeter,
            'bbox': (x, y, w, h),
            'min_rect': rect,
            'circularity': circularity
        }
```

```python
                       window_size: int = 5) -> List:
        """
        Harris角点亚像素精化

        Args:
            corners: 初始角点坐标 (N, 2)
            gray_image: 灰度图像
            window_size: 搜索窗口大小

        Returns:
            refined: 精化后的角点坐标
        """
        if len(corners) == 0:
            return []

        corners_float = corners.astype(np.float32)

        # 亚像素精化
        criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_COUNT, 30, 0.1)
        refined = cv2.cornerSubPix(
            gray_image,
            corners_float,
            (window_size, window_size),
            (-1, -1),
            criteria
        )

        return refined.tolist() if refined is not None else corners.tolist()

    def refine_edges(self, edge_points: List[Tuple[int, int]],
                    gray_image: np.ndarray) -> List[Tuple[float, float]]:
        """
        边缘点亚像素精化

        Args:
            edge_points: 边缘点列表 [(x, y), ...]
            gray_image: 灰度图像

        Returns:
            refined: 精化后的边缘点 [(x, y), ...]
        """
        if not edge_points:
            return []

        refined = []

        for y, x in edge_points:
            # 在边缘法线方向进行抛物线拟合
            refined_point = self._parabolic_fitting(y, x, gray_image)
            refined.append(refined_point)

        return refined

    def _parabolic_fitting(self, y: int, x: int,
                          gray_image: np.ndarray) -> Tuple[float, float]:
        """
        抛物线拟合亚像素边缘

        原理：在边缘法线方向采样灰度值，用抛物线拟合顶点作为亚像素位置

        Args:
            y, x: 初始边缘点坐标
            gray_image: 灰度图像

        Returns:
            sub_x, sub_y: 亚像素坐标
        """
        h, w = gray_image.shape

        # 在水平方向拟合（假设是垂直边缘）
        if y > 0 and y < h - 1:
            i0 = int(gray_image[y-1, x])
            i1 = int(gray_image[y, x])
            i2 = int(gray_image[y+1, x])

            if i0 + i2 - 2*i1 != 0:
                delta = 0.5 * (i0 - i2) / (i0 + i2 - 2*i1)
            else:
                delta = 0
        else:
            delta = 0

        return (float(x), float(y) + delta)

    def edge_line_fitting(self, edge_points: List[Tuple[float, float]]
                         ) -> Optional[dict]:
        """
        边缘直线拟合（最小二乘法）

        Args:
            edge_points: 边缘点列表

        Returns:
            line_params: 直线参数 {'angle', 'distance', 'point'}
        """
        if len(edge_points) < 2:
            return None

        points = np.array(edge_points, dtype=np.float32)

        # 使用OpenCV fitLine
        try:
            [vx, vy, x0, y0] = cv2.fitLine(
                points,
                cv2.DIST_L2,
                0,  # 距离类型
                0.01, 0.01  # 后续参数
            )

            # 计算直线参数
            angle = np.arctan2(vy, vx)[0] * 180 / np.pi
            cos_a = np.cos(np.arctan2(vy, vx)[0])
            sin_a = np.sin(np.arctan2(vy, vx)[0])
            distance = abs(x0 * sin_a - y0 * cos_a)

            return {
                'angle': float(angle),
                'distance': float(distance),
                'point': (float(x0), float(y0)),
                'direction': (float(vx), float(vy))
            }
        except:
            return None

    def detect_edge_centerline(self, binary_edge: np.ndarray) -> List:
        """
        检测边缘中心线

        Args:
            binary_edge: 二值边缘图

        Returns:
            centerline_points: 中心线点列表
        """
        # 骨架提取
        skeleton = self._skeletonize(binary_edge)

        # 查找骨架点
        points = np.where(skeleton > 0)
        centerline_points = list(zip(points[1], points[0]))  # (x, y)

        # 亚像素精化
        refined = self.refine_edges(centerline_points, binary_edge)

        return refined

    def _skeletonize(self, binary_image: np.ndarray) -> np.ndarray:
        """骨架提取"""
        skeleton = np.zeros_like(binary_image)
        element = cv2.getStructuringElement(cv2.MORPH_CROSS, (3, 3))

        temp = binary_image.copy()
        while True:
            eroded = cv2.erode(temp, element)
            opened = cv2.dilate(eroded, element)
            temp_diff = cv2.subtract(temp, opened)
            skeleton = cv2.bitwise_or(skeleton, temp_diff)
            temp = eroded.copy()

            if cv2.countNonZero(temp) == 0:
                break

        return skeleton
```

```python
# src/grasp/rrt_planner.py
import numpy as np
import random
from typing import List, Tuple, Optional, Callable
import heapq

class RRTStarNode:
    """RRT*节点"""

    def __init__(self, state: np.ndarray, parent: 'RRRStarNode' = None):
        self.state = state
        self.parent = parent
        self.cost = 0.0 if parent is None else parent.cost + np.linalg.norm(state - parent.state)

class RRTStarPlanner:
    """RRT*路径规划器"""

    def __init__(self, workspace_bounds: List[Tuple[float, float]],
                 step_size: float = 20.0,
                 max_iterations: int = 5000,
                 rewire_radius: float = 50.0,
                 goal_bias: float = 0.1):
        """
        Args:
            workspace_bounds: 工作空间边界 [(x_min, x_max), (y_min, y_max), (z_min, z_max)]
            step_size: 扩展步长
            max_iterations: 最大迭代次数
            rewire_radius: 重连半径
            goal_bias: 目标偏向概率
        """
        self.workspace_bounds = workspace_bounds
        self.step_size = step_size
        self.max_iterations = max_iterations
        self.rewire_radius = rewire_radius
        self.goal_bias = goal_bias

        self.nodes: List[RRTStarNode] = []
        self.node_index_map = {}  # 用于快速查找

    def plan(self, start: List[float], goal: List[float],
             obstacle_check_fn: Optional[Callable] = None) -> Optional[List[List[float]]]:
        """
        路径规划

        Args:
            start: 起点 [x, y, z]
            goal: 目标点 [x, y, z]
            obstacle_check_fn: 碰撞检测函数，接受state返回bool

        Returns:
            path: 路径点列表 或 None
        """
        start = np.array(start)
        goal = np.array(goal)

        # 创建起始节点
        start_node = RRTStarNode(start)
        self.nodes = [start_node]
        self.node_index_map = {tuple(start): 0}

        for i in range(self.max_iterations):
            # 采样
            if random.random() < self.goal_bias:
                random_state = goal.copy()
            else:
                random_state = self._sample_state()

            # 找最近节点
            nearest_idx, nearest_node = self._find_nearest(random_state)

            # 扩展
            new_node = self._steer(nearest_node, random_state)

            # 碰撞检测
            if obstacle_check_fn is not None and obstacle_check_fn(new_node.state):
                continue

            # 找最佳父节点
            best_parent, best_idx = self._find_best_parent(new_node, obstacle_check_fn)

            if best_parent is None:
                continue

            new_node.parent = best_parent
            new_node.cost = best_parent.cost + np.linalg.norm(new_node.state - best_parent.state)

            new_idx = len(self.nodes)
            self.nodes.append(new_node)
            self.node_index_map[tuple(new_node.state.round(2))] = new_idx

            # 重连
            self._rewire(new_node, new_idx, obstacle_check_fn)

            # 检查是否到达目标
            if np.linalg.norm(new_node.state - goal) < self.step_size:
                return self._extract_path(new_node)

        return None

    def _sample_state(self) -> np.ndarray:
        """采样随机状态"""
        state = []
        for low, high in self.workspace_bounds:
            state.append(random.uniform(low, high))
        return np.array(state)

    def _find_nearest(self, state: np.ndarray) -> Tuple[int, RRTStarNode]:
        """找最近节点"""
        min_dist = float('inf')
        nearest_idx = None
        nearest = None

        for i, node in enumerate(self.nodes):
            dist = np.linalg.norm(node.state - state)
            if dist < min_dist:
                min_dist = dist
                nearest_idx = i
                nearest = node

        return nearest_idx, nearest

    def _steer(self, from_node: RRTStarNode, to_state: np.ndarray) -> RRTStarNode:
        """扩展节点"""
        direction = to_state - from_node.state
        distance = np.linalg.norm(direction)

        if distance <= self.step_size:
            new_state = to_state
        else:
            new_state = from_node.state + (direction / distance) * self.step_size

        return RRTStarNode(new_state)

    def _find_best_parent(self, new_node: RRTStarNode,
                         obstacle_check_fn: Optional[Callable]) -> Tuple[Optional[RRTStarNode], Optional[int]]:
        """找最佳父节点"""
        best_parent = None
        best_cost = float('inf')
        best_idx = None

        for i, node in enumerate(self.nodes):
            dist = np.linalg.norm(new_node.state - node.state)
            if dist > self.step_size:
                continue

            # 检查路径是否有效
            if obstacle_check_fn is not None:
                if not self._is_edge_valid(node.state, new_node.state, obstacle_check_fn):
                    continue

            cost = node.cost + dist
            if cost < best_cost:
                best_cost = cost
                best_parent = node
                best_idx = i

        return best_parent, best_idx

    def _is_edge_valid(self, state1: np.ndarray, state2: np.ndarray,
                      obstacle_check_fn: Callable) -> bool:
        """检查边是否有效"""
        num_checks = 10
        for t in np.linspace(0, 1, num_checks):
            point = (1 - t) * state1 + t * state2
            if obstacle_check_fn(point):
                return False
        return True

    def _rewire(self, new_node: RRTStarNode, new_idx: int,
               obstacle_check_fn: Optional[Callable]):
        """重连"""
        for i, node in enumerate(self.nodes):
            if i == new_idx:
                continue

            dist = np.linalg.norm(new_node.state - node.state)
            if dist > self.rewire_radius:
                continue

            if not self._is_edge_valid(new_node.state, node.state, obstacle_check_fn):
                continue

            new_cost = new_node.cost + dist
            if new_cost < node.cost:
                node.parent = new_node
                node.cost = new_cost

                # 更新后代代价
                self._update_descendant_costs(node, new_node.cost - node.cost)

    def _update_descendant_costs(self, node: RRTStarNode, cost_delta: float):
        """更新后代节点代价"""
        # BFS更新
        queue = [node]
        visited = set()

        while queue:
            current = queue.pop(0)
            if id(current) in visited:
                continue
            visited.add(id(current))

            for child in self.nodes:
                if child.parent == current:
                    child.cost += cost_delta
                    queue.append(child)

    def _extract_path(self, end_node: RRTStarNode) -> List[List[float]]:
        """提取路径"""
        path = []
        node = end_node
        while node is not None:
            path.append(node.state.tolist())
            node = node.parent
        path.reverse()
        return path
```

```python
# src/grasp/path_smoother.py
import numpy as np
from scipy.interpolate import splprep, splev
from typing import List, Tuple

class PathSmoother:
    """路径平滑器"""

    def __init__(self):
        pass

    def b_spline_smooth(self, path: List[List[float]],
                        num_points: int = None) -> List[List[float]]:
        """
        B样条曲线平滑

        Args:
            path: 原始路径
            num_points: 输出点数

        Returns:
            smoothed: 平滑后的路径
        """
        if len(path) < 4:
            return path

        path = np.array(path)
        if num_points is None:
            num_points = len(path) * 3

        # 参数化
        t = np.linspace(0, 1, len(path))
        t_smooth = np.linspace(0, 1, num_points)

        # 对每个维度进行样条插值
        smoothed = np.zeros((num_points, path.shape[1]))
        for i in range(path.shape[1]):
            k = 3 if len(path) > 3 else len(path) - 1
            tck, _ = splprep([path[:, i], t], s=0, k=k)
            smoothed[:, i] = splev(t_smooth, tck)[0]

        # 最后一维用线性插值
        smoothed[:, -1] = np.interp(t_smooth, t, path[:, -1])

        return smoothed.tolist()

    def gradient_descent_smooth(self, path: List[List[float]],
                               obstacles: List = None,
                               iterations: int = 100,
                               step_size: float = 0.1) -> List[List[float]]:
        """
        梯度下降法平滑

        Args:
            path: 原始路径
            obstacles: 障碍物列表
            iterations: 迭代次数
            step_size: 步长

        Returns:
            smoothed: 平滑后的路径
        """
        smoothed = np.array(path, dtype=np.float64)

        for _ in range(iterations):
            for i in range(1, len(smoothed) - 1):
                # 平滑梯度
                smooth_grad = smoothed[i-1] + smoothed[i+1] - 2 * smoothed[i]

                # 障碍物梯度
                if obstacles:
                    obs_grad = self._obstacle_gradient(smoothed[i], obstacles)
                else:
                    obs_grad = 0

                # 更新
                smoothed[i] += step_size * (smooth_grad - 0.1 * obs_grad)

        return smoothed.tolist()

    def _obstacle_gradient(self, point: np.ndarray,
                          obstacles: List, epsilon: float = 1e-6) -> np.ndarray:
        """计算障碍物梯度"""
        grad = np.zeros_like(point)
        for obs in obstacles:
            if 'center' in obs:
                center = np.array(obs['center'])
                radius = obs.get('radius', 50)
                dist = np.linalg.norm(point - center)
                if dist < radius * 2:
                    grad += (point - center) / (dist + epsilon)
        return grad

    def shortcut_smoothing(self, path: List[List[float]],
                          obstacles: List = None,
                          iterations: int = 50) -> List[List[float]]:
        """
        快捷路径平滑

        Args:
            path: 原始路径
            obstacles: 障碍物列表
            iterations: 迭代次数

        Returns:
            smoothed: 平滑后的路径
        """
        smoothed = [p.copy() for p in path]

        for _ in range(iterations):
            if len(smoothed) <= 2:
                break

            # 随机选择两个不相邻的点
            i = random.randint(0, len(smoothed) - 2)
            j = random.randint(i + 2, len(smoothed) - 1)

            # 检查捷径是否可行
            if obstacles is None or self._is_shortcut_valid(smoothed[i], smoothed[j], obstacles):
                # 删除中间点
                smoothed = smoothed[:i+1] + smoothed[j:]

        return smoothed

    def _is_shortcut_valid(self, p1: List[float], p2: List[float],
                           obstacles: List) -> bool:
        """检查捷径是否无碰撞"""
        num_checks = 10
        p1 = np.array(p1)
        p2 = np.array(p2)

        for t in np.linspace(0, 1, num_checks):
            point = (1 - t) * p1 + t * p2
            for obs in obstacles:
                if 'center' in obs:
                    center = np.array(obs['center'])
                    radius = obs.get('radius', 50)
                    if np.linalg.norm(point - center) < radius:
                        return False
        return True
```

```python
from pymodbus.client import ModbusTcpClient
from pymodbus.exceptions import ModbusException
from loguru import logger
from typing import Optional, Dict, Any

class PLCCommunicator:
    """PLC通信"""

    def __init__(self, ip: str = '192.168.1.100', port: int = 502, slave_id: int = 1):
        """
        Args:
            ip: PLC IP地址
            port: 端口
            slave_id: 从站ID
        """
        self.ip = ip
        self.port = port
        self.slave_id = slave_id
        self.client: Optional[ModbusTcpClient] = None
        self.connected = False

    def connect(self) -> bool:
        """连接PLC"""
        try:
            self.client = ModbusTcpClient(self.ip, port=self.port)
            self.connected = self.client.connect()
            if self.connected:
                logger.info(f"PLC连接成功: {self.ip}:{self.port}")
            else:
                logger.error(f"PLC连接失败: {self.ip}:{self.port}")
            return self.connected
        except Exception as e:
            logger.error(f"PLC连接异常: {e}")
            return False

    def disconnect(self):
        """断开连接"""
        if self.client:
            self.client.close()
            self.connected = False
            logger.info("PLC已断开")

    def read_holding_registers(self, address: int, count: int = 1) -> Optional[list]:
        """读取保持寄存器"""
        if not self.connected:
            return None

        try:
            response = self.client.read_holding_registers(address, count, unit=self.slave_id)
            if hasattr(response, 'registers'):
                return response.registers
            return None
        except ModbusException as e:
            logger.error(f"读取寄存器错误: {e}")
            return None

    def write_register(self, address: int, value: int) -> bool:
        """写入单个寄存器"""
        if not self.connected:
            return False

        try:
            self.client.write_register(address, value, unit=self.slave_id)
            return True
        except ModbusException as e:
            logger.error(f"写入寄存器错误: {e}")
            return False

    def write_registers(self, address: int, values: list) -> bool:
        """写入多个寄存器"""
        if not self.connected:
            return False

        try:
            self.client.write_registers(address, values, unit=self.slave_id)
            return True
        except ModbusException as e:
            logger.error(f"写入寄存器错误: {e}")
            return False

    def read_defect_status(self, registers: dict) -> Optional[Dict[str, Any]]:
        """读取缺陷状态"""
        status_base = registers.get('defect_status', 40001)
        count_base = registers.get('defect_count', 40002)

        status = self.read_holding_registers(status_base, 10)
        if status:
            return {
                'defect_count': status[0],
                'edge_quality': status[1],
                'system_status': status[2],
                'error_code': status[3]
            }
        return None

    def write_grasp_command(self, x: float, y: float, z: float,
                          angle: float, registers: dict) -> bool:
        """写入抓取命令"""
        cmd_base = registers.get('grasp_command', 40100)

        # 转换坐标
        x_int = int(x * 10)   # mm * 10
        y_int = int(y * 10)
        z_int = int(z * 10)
        angle_int = int(angle * 10)

        # 写入寄存器
        success = True
        success = success and self.write_register(cmd_base, x_int)
        success = success and self.write_register(cmd_base + 1, y_int)
        success = success and self.write_register(cmd_base + 2, z_int)
        success = success and self.write_register(cmd_base + 3, angle_int)
        success = success and self.write_register(cmd_base + 4, 1)  # 执行命令

        return success

    def set_system_ready(self, ready: bool, registers: dict) -> bool:
        """设置系统就绪状态"""
        ready_reg = registers.get('system_ready', 40110)
        return self.write_register(ready_reg, 1 if ready else 0)

    def __enter__(self):
        self.connect()
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        self.disconnect()
```

```python
# src/utils/monitor.py
import psutil
import time
import torch
from typing import Dict, Any
from dataclasses import dataclass, asdict
from loguru import logger

@dataclass
class PerformanceMetrics:
    """性能指标"""
    fps: float = 0.0
    cpu_percent: float = 0.0
    memory_used_gb: float = 0.0
    memory_percent: float = 0.0
    gpu_memory_used_gb: float = 0.0
    gpu_utilization: float = 0.0
    detection_time_ms: float = 0.0
    edge_time_ms: float = 0.0
    total_time_ms: float = 0.0

class PerformanceMonitor:
    """性能监控器"""

    def __init__(self):
        self.metrics = PerformanceMetrics()
        self.frame_count = 0
        self.start_time = time.time()
        self.last_time = time.time()

    def update(self, detection_time_ms: float = 0, edge_time_ms: float = 0):
        """更新性能指标"""
        current_time = time.time()

        # 计算FPS
        self.frame_count += 1
        elapsed = current_time - self.start_time
        if elapsed > 0:
            self.metrics.fps = self.frame_count / elapsed

        # CPU和内存
        self.metrics.cpu_percent = psutil.cpu_percent(interval=0)
        mem = psutil.virtual_memory()
        self.metrics.memory_used_gb = mem.used / 1024**3
        self.metrics.memory_percent = mem.percent

        # GPU
        if torch.cuda.is_available():
            self.metrics.gpu_memory_used_gb = torch.cuda.memory_allocated() / 1024**3
            self.metrics.gpu_utilization = torch.cuda.utilization()

        # 处理时间
        self.metrics.detection_time_ms = detection_time_ms
        self.metrics.edge_time_ms = edge_time_ms
        self.metrics.total_time_ms = detection_time_ms + edge_time_ms

        self.last_time = current_time

    def get_metrics(self) -> Dict[str, Any]:
        """获取当前指标"""
        return asdict(self.metrics)

    def log_metrics(self):
        """记录性能指标"""
        m = self.metrics
        logger.debug(
            f"FPS: {m.fps:.1f} | "
            f"CPU: {m.cpu_percent:.1f}% | "
            f"Mem: {m.memory_used_gb:.2f}GB ({m.memory_percent:.1f}%) | "
            f"GPU Mem: {m.gpu_memory_used_gb:.2f}GB | "
            f"Detect: {m.detection_time_ms:.1f}ms | "
            f"Edge: {m.edge_time_ms:.1f}ms"
        )

    def reset(self):
        """重置计数器"""
        self.frame_count = 0
        self.start_time = time.time()
```

```python
# scripts/download_dataset.py
import os
import requests
import zipfile
from pathlib import Path
from typing import Dict, Optional
from loguru import logger

class DatasetDownloader:
    """数据集下载工具"""

    DATASETS: Dict[str, Dict] = {
        'NEU-DET': {
            'url': 'https://github.com/lxmic/Mamba-YOLO/releases/download/v0.0.1/NEU-DET.zip',
            'path': 'datasets/NEU-DET',
            'description': '东北大学钢板缺陷数据集'
        },
        'GC10-DET': {
            'url': 'https://github.com/lxmic/Mamba-YOLO/releases/download/v0.0.1/GC10-DET.zip',
            'path': 'datasets/GC10-DET',
            'description': 'GC10工业缺陷数据集'
        }
    }

    def __init__(self, save_dir: str = 'datasets'):
        self.save_dir = Path(save_dir)
        self.save_dir.mkdir(parents=True, exist_ok=True)

    def download(self, name: str, force: bool = False) -> Optional[Path]:
        """
        下载数据集

        Args:
            name: 数据集名称
            force: 是否强制重新下载

        Returns:
            数据集路径 或 None
        """
        if name not in self.DATASETS:
            logger.error(f"未知数据集: {name}")
            return None

        info = self.DATASETS[name]
        save_path = self.save_dir / info['path']

        if save_path.exists() and not force:
            logger.info(f"数据集已存在: {save_path}")
            return save_path

        # 下载
        zip_path = self.save_dir / f"{name}.zip"
        logger.info(f"下载 {name}...")

        try:
            response = requests.get(info['url'], stream=True, timeout=300)
            response.raise_for_status()

            total_size = int(response.headers.get('content-length', 0))
            downloaded = 0

            with open(zip_path, 'wb') as f:
                for chunk in response.iter_content(chunk_size=8192):
                    if chunk:
                        f.write(chunk)
                        downloaded += len(chunk)
                        if total_size > 0:
                            progress = (downloaded / total_size) * 100
                            if downloaded % (1024 * 1024) == 0:  # 每MB显示一次
                                logger.info(f"下载进度: {progress:.1f}%")

            # 解压
            logger.info(f"解压 {name}...")
            with zipfile.ZipFile(zip_path, 'r') as zip_ref:
                zip_ref.extractall(self.save_dir)

            # 删除zip
            os.remove(zip_path)

            logger.info(f"数据集已保存: {save_path}")
            return save_path

        except Exception as e:
            logger.error(f"下载失败: {e}")
            if zip_path.exists():
                os.remove(zip_path)
            return None

    def download_all(self):
        """下载所有数据集"""
        for name in self.DATASETS:
            self.download(name)

if __name__ == '__main__':
    downloader = DatasetDownloader()
    downloader.download_all()
```
